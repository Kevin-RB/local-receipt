receipt-app DB schema

erDiagram
    user {
        text id PK
        text email UK "NOT NULL"
        text name "NOT NULL"
        boolean emailVerified "NOT NULL default false"
        text image
        timestamptz created_at "NOT NULL default now()"
        timestamptz updated_at "NOT NULL auto-update"
    }

    session {
        text id PK
        text user_id FK "NOT NULL"
        text token UK "NOT NULL"
        timestamptz expires_at "NOT NULL"
        text ip_address
        text user_agent
        timestamptz created_at "NOT NULL default now()"
        timestamptz updated_at "NOT NULL auto-update"
    }

    account {
        text id PK
        text user_id FK "NOT NULL"
        text issuer "NOT NULL"
        text account_id "NOT NULL"
        text password
        text access_token
        timestamptz access_token_expires_at
        text refresh_token
        timestamptz refresh_token_expires_at
        text id_token
        text scope
        timestamptz created_at "NOT NULL default now()"
        timestamptz updated_at "NOT NULL auto-update"
    }

    verification {
        text id PK
        text identifier "NOT NULL"
        text value "NOT NULL"
        timestamptz expires_at "NOT NULL"
        timestamptz created_at "NOT NULL default now()"
        timestamptz updated_at "NOT NULL default now()"
    }

    receipts {
        uuid id PK "default gen_random_uuid()"
        text user_id FK "NOT NULL indexed"
        text status "NOT NULL default pending (uploading|pending|processing|done|error)"
        text minio_object_key
        text merchant_name
        text merchant_abn
        text merchant_address
        text merchant_store_id
        text receipt_number
        timestamptz transaction_datetime
        numeric_10_2 subtotal
        numeric_10_2 gst
        numeric_10_2 total
        text payment_method "(cash|card|other)"
        boolean has_integrity_warning "NOT NULL default false"
        timestamptz created_at "NOT NULL default now() indexed"
    }

    receipt_items {
        uuid id PK "default gen_random_uuid()"
        uuid receipt_id FK "NOT NULL"
        text name "NOT NULL"
        numeric quantity
        numeric_10_2 unit_price
        numeric_10_2 line_total "NOT NULL negative = discount"
    }

    user ||--o{ session : has
    user ||--o{ account : has
    user ||--o{ receipts : owns
    receipts ||--o{ receipt_items : contains
